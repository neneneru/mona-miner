cmake_minimum_required(VERSION 3.26)
project(mona_v2_isolated_prototype LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "64-bit host required")
endif()
option(MONA2_SANITIZE "CPU ASan/UBSan authoring build" OFF)
option(MONA2_FROZEN_GPU "Compile host-only frozen-image GPU adapter" ${WIN32})
find_package(Threads REQUIRED)
file(GLOB JANSSON_FILES CONFIGURE_DEPENDS vendor/jansson/*.c)
add_library(jansson STATIC ${JANSSON_FILES})
target_include_directories(jansson PUBLIC vendor/jansson)
target_compile_definitions(jansson PRIVATE _CRT_SECURE_NO_WARNINGS)
add_library(reference STATIC
 vendor/cpu_upstream/sph/blake.c vendor/cpu_upstream/sph/keccak.c
 vendor/cpu_upstream/sph/cubehash.c vendor/cpu_upstream/sph/skein.c
 vendor/cpu_upstream/sph/bmw.c vendor/cpu_upstream/sph/sph_sha2.c
 vendor/cpu_upstream/lyra2/Lyra2.c vendor/cpu_upstream/lyra2/Sponge.c)
target_include_directories(reference PUBLIC vendor/cpu_upstream)
target_compile_definitions(reference PRIVATE _CRT_SECURE_NO_WARNINGS)
add_library(mona2_core STATIC src/base.cpp src/json.cpp src/session.cpp src/accounting.cpp src/protocol.cpp src/cli.cpp)
target_include_directories(mona2_core PUBLIC include)
target_link_libraries(mona2_core PUBLIC jansson reference Threads::Threads)
add_library(mona2_io STATIC src/transport.cpp src/actor.cpp src/worker.cpp)
target_link_libraries(mona2_io PUBLIC mona2_core)
if(WIN32)
 target_compile_definitions(mona2_io PRIVATE _WIN32_WINNT=0x0A00 UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_link_libraries(mona2_io PUBLIC Ws2_32)
endif()
foreach(target mona2_core mona2_io)
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8)
  target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
 else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
 endif()
endforeach()
if(MONA2_SANITIZE AND NOT MSVC)
 add_compile_options(-fsanitize=address,undefined,float-cast-overflow -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined,float-cast-overflow)
 foreach(t mona2_core mona2_io jansson reference)
  target_compile_options(${t} PRIVATE -fsanitize=address,undefined,float-cast-overflow -fno-omit-frame-pointer)
 endforeach()
endif()
enable_testing()

add_test(NAME protocol_tests COMMAND protocol_tests)

add_test(NAME console_tests COMMAND console_tests)
add_executable(loopback_probe tests/loopback_probe.cpp)
target_link_libraries(loopback_probe PRIVATE mona2_io)
add_executable(cli_probe tests/cli_probe.cpp)
target_link_libraries(cli_probe PRIVATE mona2_io)
if(MONA2_FROZEN_GPU)
 if(NOT WIN32)
  message(FATAL_ERROR "Frozen GPU host only built on Windows")
 endif()
 if(NOT MONA2_CUDA_ROOT)
  message(FATAL_ERROR "Pinned CUDA SDK root required; no nvcc discovery permitted")
 endif()
 add_library(CUDA::cuda_driver STATIC IMPORTED)
 set_target_properties(CUDA::cuda_driver PROPERTIES IMPORTED_LOCATION "${MONA2_CUDA_ROOT}/lib/x64/cuda.lib" INTERFACE_INCLUDE_DIRECTORIES "${MONA2_CUDA_ROOT}/include")
 if(NOT DEFINED MONA2_IMAGE_OBJECT OR NOT DEFINED MONA2_GENERATED_INCLUDE)
  message(FATAL_ERROR "Set MONA2_IMAGE_OBJECT and MONA2_GENERATED_INCLUDE; see ../build-tools/README.md")
 endif()
 add_library(mona2_gpu STATIC src/frozen_backend.cpp)
 target_include_directories(mona2_gpu PUBLIC "${MONA2_GENERATED_INCLUDE}")
 target_link_libraries(mona2_gpu PUBLIC mona2_core CUDA::cuda_driver)
 add_executable(mona-miner-prototype src/main.cpp src/application.cpp src/developer_destination.cpp "${MONA2_IMAGE_OBJECT}")
 target_link_libraries(mona-miner-prototype PRIVATE mona2_gpu mona2_io)
 add_executable(live_probe tests/live_probe.cpp src/application.cpp src/developer_destination.cpp "${MONA2_IMAGE_OBJECT}")
 target_link_libraries(live_probe PRIVATE mona2_gpu mona2_io)
 if(MSVC)
  foreach(e mona-miner-prototype live_probe)
   target_link_libraries(${e} PRIVATE delayimp)
   target_link_options(${e} PRIVATE /DELAYLOAD:nvcuda.dll)
  endforeach()
 endif()
 add_executable(product_gpu_probe tests/product_gpu_probe.cpp "${MONA2_IMAGE_OBJECT}")
 target_link_libraries(product_gpu_probe PRIVATE mona2_gpu mona2_io)
 add_executable(gpu_local_tests tests/gpu_local_tests.cpp "${MONA2_IMAGE_OBJECT}")
 target_link_libraries(gpu_local_tests PRIVATE mona2_gpu mona2_io)
endif()
